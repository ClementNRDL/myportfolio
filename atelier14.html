Quel est le résultat de OR ?

Qu’est-ce que le code ci-dessous va sortir ?

L’opérateur || retourne la première valeur vraie rencontrée. Ici, le résultat affiché est 2. 

alert( null || 2 || undefined );

Quel est le résultat des alertes OR ?

Qu’est-ce que le code ci-dessous va sortir ?

alert(1) s’exécute en premier et affiche 1, puis retourne undefined. L’opérateur || continue et exécute alert(2) qui affiche 2. alert(3) n’est jamais exécuté. 

alert( alert(1) || 2 || alert(3) );

Quel est le résultat de AND ?

Qu’est-ce que ce code va afficher ?

L’opérateur && retourne la première valeur fausse rencontrée. Ici, le résultat affiché est null. 

alert(1 && null && 2);

Quel est le résultat des alertes AND ?

Qu’est-ce que ce code va afficher ?

alert(1) s’exécute et affiche 1, puis retourne undefined. L’opérateur && s’arrête, donc alert(2) n’est jamais exécuté. 

alert( alert(1) && alert(2) );

Le résultat de OR AND OR

    Créer un calendrier

    Générer dynamiquement un calendrier mensuel à l’aide de JavaScript. Le calendrier est créé sous forme de tableau HTML. 

    function createCalendar(elem, year, month) {
    let table = document.createElement("table");

    let days = ["MO", "TU", "WE", "TH", "FR", "SA", "SU"];
    let tr = document.createElement("tr");

    for (let day of days) {
        let th = document.createElement("th");
        th.textContent = day;
        tr.append(th);
    }
    table.append(tr);

    let date = new Date(year, month - 1, 1);
    let startDay = (date.getDay() + 6) % 7;

    tr = document.createElement("tr");
    for (let i = 0; i < startDay; i++) {
        tr.append(document.createElement("td"));
    }

    while (date.getMonth() === month - 1) {
        let td = document.createElement("td");
        td.textContent = date.getDate();
        tr.append(td);

        if ((date.getDay() + 6) % 7 === 6) {
            table.append(tr);
            tr = document.createElement("tr");
        }

        date.setDate(date.getDate() + 1);
    }

    if (tr.children.length > 0) {
        table.append(tr);
    }

    elem.append(table);
}

let timerId;

/* Fonction qui met à jour l’heure */
function updateClock() {
    let now = new Date();

    let h = String(now.getHours()).padStart(2, "0");
    let m = String(now.getMinutes()).padStart(2, "0");
    let s = String(now.getSeconds()).padStart(2, "0");

    document.querySelector(".seconds").textContent = s;

/* Fonction pour démarrer l’horloge */
function startClock() {
    if (!timerId) {
        updateClock();
        timerId = setInterval(updateClock, 1000);
    }
}

/* Fonction pour arrêter l’horloge */
function stopClock() {
    clearInterval(timerId);
    timerId = null;
}

    Déplacez le ballon à travers le terrain

    Cliquez dans le terrain pour déplacer le ballon.

    /* ================= EXERCICE 52 ================= */
/* Déplacer le ballon à travers le terrain */

function initBallMove() {
    const field = document.getElementById("field");
    const ball = document.getElementById("ball");

    field.onclick = function (event) {

        const rect = field.getBoundingClientRect();

        let x = event.clientX - rect.left - ball.offsetWidth / 2;
        let y = event.clientY - rect.top - ball.offsetHeight / 2;

        x = Math.max(0, Math.min(x, field.clientWidth - ball.offsetWidth));
        y = Math.max(0, Math.min(y, field.clientHeight - ball.offsetHeight));

        ball.style.left = x + "px";
        ball.style.top = y + "px";
    };
}

    Info-bulle avec data-tooltip

    Afficher une info-bulle au survol d’un élément possédant l’attribut data-tooltip. L’info-bulle la plus imbriquée est prioritaire. 

    /* ================= EXERCICE 47 ================= */
/* Info-bulle avec data-tooltip */

let tooltip;

document.addEventListener("mouseover", function (event) {
    let target = event.target.closest("[data-tooltip]");
    if (!target) return;

    tooltip = document.createElement("div");
    tooltip.className = "tooltip";
    tooltip.textContent = target.dataset.tooltip;

    document.body.append(tooltip);

    let rect = target.getBoundingClientRect();
    tooltip.style.left = rect.left + "px";
    tooltip.style.top = rect.top - tooltip.offsetHeight - 5 + "px";
});

document.addEventListener("mouseout", function () {
    if (tooltip) {
        tooltip.remove();
        tooltip = null;
    }
});
